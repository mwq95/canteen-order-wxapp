## 四、云函数实现

### 4.1 quickstartFunctions 新增方法

```javascript
// 验证员工身份
const verifyStaff = async (event) => {
  const { phone } = event;
  try {
    const res = await db.collection('staffs').where({
      phone: String(phone),
      status: 'active'
    }).get();
    if (res.data.length > 0) {
      return { success: true, staff: res.data[0] };
    }
    return { success: false, error: '未找到该员工信息' };
  } catch (e) {
    return { success: false, error: e.message };
  }
};

// 获取用户信息
const getUserInfo = async (event, openid) => {
  try {
    const res = await db.collection('users').where({ _openid: openid }).get();
    if (res.data.length > 0) {
      const user = res.data[0];
      return {
        isVerified: user.isVerified || false,
        role: user.role || 'staff',
        phone: user.phone || null,
        name: user.name || null,
        userInfo: user.userInfo || null
      };
    }
    return { isVerified: false, role: 'staff', phone: null, name: null, userInfo: null };
  } catch (e) {
    return { isVerified: false, role: 'staff', phone: null, name: null, userInfo: null };
  }
};

// 创建或更新用户
const createOrUpdateUser = async (event, openid) => {
  const { phone, name, role } = event;
  try {
    const userRes = await db.collection('users').where({ _openid: openid }).get();
    const userData = {
      phone,
      name,
      isVerified: true,
      role: role || 'staff',
      updateTime: new Date()
    };
    if (userRes.data.length > 0) {
      await db.collection('users').doc(userRes.data[0]._id).update({ data: userData });
    } else {
      userData.createTime = new Date();
      await db.collection('users').add({ data: { ...userData, _openid: openid } });
    }
    return { success: true };
  } catch (e) {
    return { success: false, error: e.message };
  }
};

// 获取用户提醒设置
const getUserSettings = async (event) => {
  const { phone } = event;
  try {
    const res = await db.collection('users').where({ phone }).get();
    if (res.data.length > 0) {
      return {
        subscribeOrderReminder: res.data[0].subscribeOrderReminder !== false,
        subscribeMealReminder: res.data[0].subscribeMealReminder !== false
      };
    }
    return { subscribeOrderReminder: true, subscribeMealReminder: true };
  } catch (e) {
    return { subscribeOrderReminder: true, subscribeMealReminder: true };
  }
};

// 更新用户提醒设置
const updateUserSettings = async (event) => {
  const { phone, subscribeOrderReminder, subscribeMealReminder } = event;
  try {
    await db.collection('users').where({ phone }).update({
      data: { subscribeOrderReminder, subscribeMealReminder, updateTime: new Date() }
    });
    return { success: true };
  } catch (e) {
    return { success: false, error: e.message };
  }
};

// 保存菜单
const saveMenu = async (event, openid) => {
  const { date, meals } = event;
  try {
    const existing = await db.collection('menus').where({ date }).get();
    if (existing.data.length > 0) {
      await db.collection('menus').doc(existing.data[0]._id).update({
        data: { meals, updateTime: new Date() }
      });
    } else {
      await db.collection('menus').add({
        data: { date, meals, createTime: new Date(), updateTime: new Date() }
      });
    }
    return { success: true };
  } catch (e) {
    return { success: false, error: e.message };
  }
};
```

### 4.2 orderFunctions 修改方法

```javascript
// 获取用户当日订单（新增）
const getTodayOrders = async (event) => {
  const { phone, date } = event;
  try {
    const mealTypes = ['早餐', '午餐', '晚餐'];
    const results = {};
    for (const mealType of mealTypes) {
      const res = await db.collection('orders').where({
        phone, date, mealType
      }).get();
      results[mealType] = res.data[0] || null;
    }
    return { success: true, data: results };
  } catch (e) {
    return { success: false, error: e.message };
  }
};

// 获取历史订单（新增）
const getHistoryOrders = async (event) => {
  const { phone } = event;
  try {
    const res = await db.collection('orders')
      .where({ phone })
      .orderBy('createTime', 'desc')
      .get();
    return { success: true, data: res.data };
  } catch (e) {
    return { success: false, error: e.message };
  }
};

// submitOrder 修改 - 确保记录 phone
const submitOrder = async (event) => {
  const wxContext = cloud.getWXContext();
  const { date, mealType, dishes, phone, existingOrderId } = event;

  const orderData = {
    date,
    mealType,
    dishes,
    phone: phone || '',
    status: 'pending',
    createTime: db.serverDate(),
    updateTime: db.serverDate(),
    _openid: wxContext.OPENID
  };

  if (existingOrderId) {
    await db.collection('orders').doc(existingOrderId).update({ data: orderData });
    return { success: true, message: '订单已更新' };
  } else {
    const res = await db.collection('orders').add({ data: orderData });
    return { success: true, message: '订餐成功', orderId: res._id };
  }
};
```

## 六、注意事项

1. **openid获取**: 云函数中通过 `cloud.getWXContext().OPENID` 获取，无需前端传递
2. **phone传递**: 前端从 `app.globalData.phone` 获取，传递给需要 phone 的云函数
3. **向后兼容**: 已有的云函数方法不要删除，确保历史数据兼容
4. **错误处理**: 云函数中要做好 try-catch，返回明确的错误信息
5. **评价功能**: evaluate.js 保持前端直查，不纳入本次改造范围
